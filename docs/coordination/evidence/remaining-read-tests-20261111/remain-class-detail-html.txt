    function PopulateDataInGrid(response, status, class_id)

        if(status=='success')

            var response_header = response.getAllResponseHeaders().toLowerCase();

            var leaderboard_UpcomingJO_Rows_Index = new Array();
            var inCountdownTrip = -989898;
            
            //After parsing filter the records in leader board and lined up grids
            for(var index=0;index<Number(obj.recs);index++)
            {
                //Set current trip if trip is gone or it is in countdown
                if(obj.curr == obj.rows[index].id && (obj.rows[index].Gone == "1" || !isEmptyValue(obj.rows[index].r1_cd))){
                    inCountdownTrip = !isEmptyValue(obj.rows[index].r1_cd) ? obj.rows[index].id : inCountdownTrip;
                    last_updated_at = obj.curr_updated_at;
                    if(last_updated_at != ""){
                        last_updated_at = last_updated_at.split(" ");
                        last_updated_date = last_updated_at[0];
                        last_updated_time = timeToSeconds(last_updated_at[1]);
                        time_elapsed = server_time_now-last_updated_time;
                        time_in_miliseconds = ((new Date())-(time_elapsed*1000));
                        dateObj = new Date(time_in_miliseconds);
                        if((last_updated_date == server_date_now) && ((time_elapsed)<(300))){ //Last update is within last 5 minutes
                            curr_date_time_valid = true;
                        }
                        
                        //Check that current trip has time and has no update for 20 secs after time populated >>>
                        if(curr_date_time_valid){
                            vb_newDataHasTimeInfo = (!isEmptyValue(obj.rows[index].R1T) || !isEmptyValue(obj.rows[index].JOT));;
                            vb_hasJO = true;
                            if(vb_newDataHasTimeInfo){ //Existing data has time
                                switch(table_type){
                                    //No Jump-off/Jump-off after delay
                                    case 0: 
                                    case 1:
                                    case 5:
                                        vb_hasJO = true;
                                        break;
                                        
                                    //Wait for Jump-Off Time
                                    case 2:
                                        vb_hasJO = true;
                                        if(isEmptyValue(obj.rows[index].R1F)){ //If clean Round 1
                                            vb_hasJO = !isEmptyValue(obj.rows[index].JOT);
                                        }
                                        break;
                                        
                                    //Wait for JO even if R1F > 0    
                                    case 4:
                                        vb_hasJO = !isEmptyValue(obj.rows[index].JOT);

                                }
                                
                                //No update after time was added for 20 seconds

                    }
                    
                    if(curr_date_time_valid){
                        //Class has live trip condition = true
                    }

                    LiveTripRows_classID.push(class_id);

                }

                //Add row in leaderboard

                vb_JOQualified = false;
                if(obj.rows[index].Gone == "1" && after_delay_jo){
                    vb_round1_clear = isEmptyValue(obj.rows[index].R1F);
                    if(table_type == 5){ //R2 trips are decided based on callbacks if callbacks are set in jumper table type = 5
                        var tripPosition = parseInt(obj.rows[index].Pos);
                        if(JOGRowsCount > 0 && tripPosition > 0 && tripPosition <= JOGRowsCount){
                            vb_round1_clear = true;
                        }
                        else {
                            vb_round1_clear = false;
                        }
                    }
                    vb_round1_clear = vb_round1_clear && !isEmptyValue(obj.rows[index].R1T);
                    vb_round1_clear = vb_round1_clear && !isEliminatedValue(obj.rows[index].R1T) && !isEliminatedValue(obj.rows[index].R1F); //Not eliminated value

                    //check Round is clean OR JO data entered then Trip is JO Qualified
                    vb_JOQualified = vb_round1_clear || !isEmptyValue(obj.rows[index].JOF) || !isEmptyValue(obj.rows[index].JOT);
                    vb_JOQualified = vb_JOQualified && !isEliminatedValue(obj.rows[index].R1T) && !isEliminatedValue(obj.rows[index].R1F); //Not eliminated value in R1
                    if(vb_JOQualified){
                        vb_hasJOQualifiedRows = true;
                        vb_JOGone = !isEmptyValue(obj.rows[index].JOT) || isEliminatedValue(obj.rows[index].JOF);

    function setPositions(class_id){
        //Set Valid positions
        var class_index = GetClassIndex(class_id);
        var copy_obj = live_class_obj.class_trips[class_index].leaderboard;
        var listData = JSON.parse(JSON.stringify(copy_obj)); //copy_obj.slice();
        var vl_position = 0;
        var last_position = -88; //invalid value
        for(var NewRowCount=0; NewRowCount<listData.length;NewRowCount++){

            //Trip is not eliminated trip
            if(!isEliminatedValue(listData[NewRowCount].Pos)){
                if(listData[NewRowCount].Pos != last_position){ //Check same position for last row
                    vl_position++;
                }

                if(after_delay_jo && listData[NewRowCount].Pos.toString().indexOf("JO")>=0 && isEmptyValue(listData[NewRowCount].JOT) && !isEliminatedValue(listData[NewRowCount].JOF)){
                    live_class_obj.class_trips[class_index].leaderboard[NewRowCount].Pos = "JO-" + vl_position; //Set position for trips not gone in JO Round
                }
                else{
                    live_class_obj.class_trips[class_index].leaderboard[NewRowCount].Pos = vl_position.toString();
                }
                
                //Update Position for JO Rows to ensure Position counter change
                if(listData[NewRowCount].Pos.toString().indexOf("JO")>=0){
                    listData[NewRowCount].Pos = listData[NewRowCount].Pos + "-" + NewRowCount; 
                }
                last_position = listData[NewRowCount].Pos;
            }
        }
    }
    
    function PopulateLinedUp(class_id){
        var trips_to_go = 0; 
        var class_index = GetClassIndex(class_id);
        live_class_obj.class_trips[class_index].lined_up = [];
        live_class_obj.next_trip = {};
        vb_upNextRowAdded = false;
        
        var lined_in_pos = 0;
        var lined_up_rows = 1;
        var lined_up_entries = new Array();
        var class_lined_up_entries = new Array();
        
        //Sort Lined up rows on OOG and Entry No.
        array_multisort(LinedUpRows_Scr, 'SORT_ASC', 'SORT_NUMERIC',
                            LinedUpRows_OOG, 'SORT_ASC', 'SORT_NUMERIC', 
                            LinedUpRows_ENo, 'SORT_ASC', 'SORT_NUMERIC',
                            LinedUpRows, 'SORT_ASC',
                            LinedUpRows_ClassID, 'SORT_ASC');
                
        
        for(var counter=0; counter < LinedUpRows.length; counter++){
            if(lined_up_entries.indexOf(LinedUpRows[counter].ENo) < 0){ //Show One trip of Multiple trips of same entry once
                lined_up_entries.push(LinedUpRows[counter].ENo);
                lined_in_pos = order_of_go_set ? lined_up_rows : "";
                lined_up_rows++;

                if (LinedUpRows[counter].Scr == "0"){
                    trips_to_go++;
                    
                    if(!vb_upNextRowAdded && order_of_go_set){ //Set UP NEXT ROW
                        live_class_obj.next_trip = LinedUpRows[counter];
                        vb_upNextRowAdded = true;
                    }
                }
            }
            
            if(LinedUpRows_ClassID[counter] == class_id && class_lined_up_entries.indexOf(LinedUpRows[counter].ENo) < 0){ //Push lined up trips of current class
                live_class_obj.class_trips[class_index].lined_up.push(LinedUpRows[counter]);
                class_lined_up_entries.push(LinedUpRows[counter].ENo);
            }
        }
        
        //If JUMP-OFF is after some delay and Some Trips JO is gone
        if(after_delay_jo && vb_hasJOGoneTrips){
            live_class_obj.next_trip = {};
            vb_upNextRowAdded = false;
        }
        
        //Set Up Next Horse for rounds whose JUMP-OFF is after some delay of 1st round
        if(table_type == 0){///Sort on Actual Order for JO Qualified trips BUT Jump-Off not gone
            array_multisort(UpcomingJO_Rows_ActOOG, 'SORT_ASC', 'SORT_NUMERIC', 
                                UpcomingJO_Rows, 'SORT_ASC', 
                                UpcomingJO_Row_IDs, 'SORT_ASC', 'SORT_NUMERIC');
        }
        else if(table_type == 5){
            array_multisort(UpcomingJO_Rows_R1Faults, 'SORT_DESC', 'SORT_NUMERIC',
                                UpcomingJO_Rows_R1Time, 'SORT_DESC', 'SORT_NUMERIC',
                                UpcomingJO_Rows_ActOOG, 'SORT_ASC', 'SORT_NUMERIC', 
                                UpcomingJO_Rows, 'SORT_ASC', 
                                UpcomingJO_Row_IDs, 'SORT_ASC', 'SORT_NUMERIC');
        }
        
        for(vl_upcomingJo_counter=0; vl_upcomingJo_counter<UpcomingJO_Rows.length; vl_upcomingJo_counter++){
            if(LiveTripID != UpcomingJO_Rows[vl_upcomingJo_counter].id){ //ignore live trip from up next if its not yet have JO Data
                if(after_delay_jo && UpcomingJO_Rows.length>0 && (trips_to_go==0 || vb_hasJOGoneTrips)){
                    live_class_obj.next_trip = UpcomingJO_Rows[vl_upcomingJo_counter];
                    vb_upNextRowAdded = true;
                    break;
                }
            }
        }
        
        
        if(!vb_upNextRowAdded){
            live_class_obj.next_trip = {};
            vb_upNextRowAdded = true;
        }
    }
    function PopulateLiveNow(){
        live_class_obj.live_trip = {};
        
        if(LiveTripRows.length > 0){
            //Sort Live Now on basis of time
            array_multisort(
                    LiveTripRows_elapsedtime, 'SORT_ASC', 'SORT_NUMERIC',
                    LiveTripRows_validtime, 'SORT_DESC',
                    LiveTripRows_classID, 'SORT_ASC',
                    LiveTripRows, 'SORT_ASC',
                    LiveTripRows_UpdateTime, 'SORT_ASC');
                  
            if(LiveTripRows_validtime[0] == true){ //Most recent trip is live
                live_class_obj.live_trip = LiveTripRows[0];
                LiveTripID = LiveTripRows[0].id;
                
                //Pyramid timing fields (pt, r1_cd, r1_start, r1_start_time, r1_done, jo_cd, jo_start, jo_start_time, jo_done, time_4d)
                curr_status = "";
                if(LiveTripRows[0].pt == true || LiveTripRows[0].pt == "1"){ //Handle Pyramid timing data
                    fromPyramid = true;
                    //Get current GMT time from JSON reponse header instead of client time
                    var current_time = getGMTDateTime(new Date(LiveTripRows_UpdateTime[0]));
                    var json_time = ""; 
                    var countdown_limit = 0;
                    //var now = new Date(); 
                    //current_time = getGMTDateTime(now);
                    
                    //Check JO Round info first
                    if(isEmptyValue(LiveTripRows[0].JOT)){
                        which_round = "JO";
                        
                        if(!isEmptyValue(LiveTripRows[0].jo_done)){
                            curr_status = "TS"; //Stop JO timer
                        }
                        else if(!isEmptyValue(LiveTripRows[0].jo_start)){ //Show Round 1 Timer
                            curr_status = "TA"; //JO timer active 
                            json_time = getGMTDateTime(new Date(LiveTripRows[0].jo_start)); 
                        }
                        else if(!isEmptyValue(LiveTripRows[0].jo_cd) && isEmptyValue(LiveTripRows[0].JOF)){ 
                            curr_status = "CDA"; //JO countdown timer active
                            json_time = getGMTDateTime(new Date(LiveTripRows[0].jo_cd));
                            countdown_limit = jo_countdown_limit;
                        }
                    }
                    
                    //Check Round 1 only if trip has No JO round info

            let sorted = 0;
            let index = 0;
            
            // Loop until sorted (-1 or 1) or until the sort keys have been processed.
            while (sorted === 0 && index < sortKeys.length) {
                const key = sortKeys[index];
                
                if (key) {
                    const direction = sortObject[key];
                    
                    sorted = keySort(a[key], b[key], direction);
                    index++;
                }
            }
        
            return sorted;
        });
    }
