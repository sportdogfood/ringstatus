var last_width = $(window).width();

function update_title()
{
  $.get('show_title.php',
    { show: show_no },
    function(data){
      $('#show_title').text(data);
  }); 
}

function resetSearch()
{
  // clear the textbox
//$('#txtSearch').val('');
  // show all table rows
  $("table").show();
  $('table tr').show();
  $('table thead').show();
  // remove any no records rows
  $('.norecords').remove();
  $("#total_pm").hide();
  // make sure we re-focus on the textbox for usability
  $('#txtSearch').focus();
}

function doSearch()
{
  if ($('#txtSearch').val().length > 2)
  {
    $("table tbody").each(function(i)
    {        
      $(this).find('tr').each(function() {
        $(this).hide();
      });
      $(this).find('tr td:containsNoCase(\'' + jQuery('#txtSearch').val() + '\')').parent().show();
     
      var n_tr = $(this).find('tr:visible').length;
      if (n_tr == 0)
      {
        $(this).parent().find("thead").hide();
        $(this).parent().hide();
      }  
      else  
        $(this).parent().find("thead").show();
    }).promise().done(function() {
      var sum = 0;
      $('tr:visible').each(function(){
        var pm =  $(this).data("pm");
        if (typeof pm !== 'undefined')
          sum = sum + pm;
      });
      if (sum > 0)
      {
        sum = 'Total Prize Money: $' + numberWithCommas(sum);
        $("#total_pm").text(sum);
        $("#total_pm").show();
      } 
      else
        $("#total_pm").hide();
   });
  }
  else if ($('#txtSearch').val().length == 0)
  {
   // if the user removed all of the text, reset the search
   resetSearch();
  }

  // if there were no matching rows, tell the user
  if ($('table tr:visible').length == 1)
  {
   // remove the norecords row if it already exists
   $('.norecords').remove();
   // add the norecords row
   $('.result').append('<tr class="norecords"><td colspan="5" class="Normal">No records were found</td></tr>');
  }
}

$.expr[":"].containsNoCase = function(el, i, m)
{
  var search = m[3];
  if (el.tagName == 'TH')
    return true;
  if (!search) return false;
    return eval("/" + search + "/i").test($(el).text());
};




function hide_all()
{
  $('#hscom_logo').show();
  $('#trainer').hide();

  $('#numerical4').hide();
  $('#classes').hide();
  $('#hrot_search').hide();

  $('.cd-top').hide();

  scroll_y = 0;
  $('#txtSearch').val('');
  search_text = '';
}

function display_classes(filter = '')
{
  $.get('srched_classes.php',
    { filter: filter },
    function(data)
    { 
      if (data == 'NO MATCH')
        $('#no_match').html('<strong>No Match</strong>');
      else
      {    
        $('#no_match').text('');
        $('#selected_classes tbody').html(data);
      }  
  }); 
}

function display_numerical()
{
  $.get('numerical4.php',{},
    function(data)
    { 
      $('#numerical4').html(data);
      $('#numerical4').show();
      nsize = '';
      $('.nlg').hide();
      $('.nmd').hide();
      $('.nsm').hide();
      numerical = true;
      resize();
      $('#txtSearch').val(search_text);
      if (search_text != '')
        doSearch();
      window.scrollTo(0, scroll_y);
      $('.cd-top').show();
    }); 
}

function get_classes(filter = '')
{
  var query = "SELECT\
    classes.number,\
    classes.name,\
    COUNT(eclasses.eclass_no) AS n_entries,\
    classes.unique_no AS class_no\
  FROM\
    classes\
  LEFT JOIN eclasses ON classes.show_no = eclasses.show_no AND classes.unique_no = eclasses.class_no\
  WHERE classes.show_no = " + show_no;
  
  if (filter != '')
    query += filter;
  query += " GROUP BY classes.number, classes.name  order by lpad(number, 7, '0')";
}

function numberWithCommas(x)
{
  return x.toString().replace(/\B(?=(\d{3})+(?!\d))/g, ",");
}

function resize()
{
  var width = $(window).width();
  if (! numerical) 
  {
    if (width <= 505 && size != 'sm')
    {
      size = 'sm';
      $('.lg').hide();
      $('.md').hide();
      $('.sm').show();
    }  
    else
    {
    	if (width > 505 && width < 700 && size != 'md')
    	{
  	    size = 'md';
  	    $('.lg').hide();
  	    $('.sm').hide();
  	    $('.md').show();
    	}
    	else
    	{
  	    if (width > 700 && size != 'lg' )
  	    {
  	      size = 'lg';
  	      $('.sm').hide();
          $('.md').hide();
  	      $('.lg').show();
  	    }
      }  
    }
  }
  else
  {
    if (width <= 605 && nsize != 'sm')
    {
      nsize = 'sm';
      $('.nlg').hide();
      $('.nmd').hide();
      $('.nsm').show();
    }  
    else
    {
    	if (width > 605 && width < 850 && nsize != 'md')
    	{
  	    nsize = 'md';
  	    $('.nlg').hide();
  	    $('.nsm').hide();
  	    $('.nmd').show();
    	}
    	else
    	{
  	    if (width > 850 && nsize != 'lg' )
  	    {
  	      nsize = 'lg';
  	      $('.nsm').hide();
          $('.nmd').hide();
  	      $('.nlg').show();
  	    }
      }  
    }
    doSearch();
  }  
  last_width = width;    
}
 
$(window).resize(function()
{
  resize();
});

function hasStoredState()
{
  return localStorage.getItem('pageState') !== null;
}

function saveState(numerical)
{
  let state = {
    numerical: numerical, 
    scroll_y: window.scrollY || window.pageYOffset,
    search: $('#txtSearch').val()
  };
  localStorage.setItem('pageState', JSON.stringify(state));
}    

function restoreState()
{
  let state = JSON.parse(localStorage.getItem('pageState'));
  if (state.numerical)
  {
    scroll_y = state.scroll_y;
    search_text = state.search;
    nsize = '';
    display_numerical();
  }  
  localStorage.removeItem('pageState');
}

  {
     filter_op = ' AND (';

       filter_op = ' OR (';    
     let srch = $('#classes_search').val();
     if (srch.length >= 2)
     {
//     $('#class_results_btn').prop("disabled",true);
       let tokens = srch.split(" ");
       let filter = '(';
       let len = tokens.length;
       for (i = 0; i < len; i++)
       {
         if (tokens[i].length > 0)
         {
           if (filter != '(')
             filter += filter_op;
           tokens[i] = tokens[i].replace("'", "\\'"); // escape any single quotes 
           filter += "cl.name like '%" + tokens[i] + "%' OR cl.number = '" + tokens[i] + "' OR ss.name = '" + tokens[i] + "')";
         }
       }
       display_classes(filter);
//     $('#class_results_btn').prop("disabled",false);
     }
     else
       display_classes();
  }
  var timer=null;
  document.getElementById("classes_search").onkeyup = ()=>{
    clearInterval(timer);
    timer = setInterval(()=>{
              do_classes_search(); 
              clearInterval(timer);
            },1000);
  };

  $('#and_or_cb').change(function()
  {
    do_classes_search();
  });

  $('#class_results_btn').click(function()
  {
    $('#results').hide();

      $('#hscom_logo').hide();

      $("#busy").hide();
